% figures04.tex -- Chapter 4 figures redrawn in TikZ.
%
% Label clearance, chapter-local (2026-08-17).  Every figure in this chapter
% is a point-on-a-line diagram whose labels sit directly over their point.
% The global `outer sep' was tightened to 2.2pt in the same pass that turned
% the point dots into fixed 3.2pt nodes -- and a 3.2pt dot reaches 1.6pt out
% of its point, so at 2.2pt the label box lands ON the dot it labels.
% check_labels bears this out exactly: the collision count for this chapter
% falls off a cliff at 3.2pt (39 -> 2) and does not improve above it.  3.2pt
% is therefore the tightest clearance that still clears the new dots, and is
% still tighter than the 3.4pt this book used before the dots existed.
% Coordinates are reconstructions; every congruence the printed figure
% asserts holds EXACTLY here (see tools/constraints/ch04.py).
% Self-contained: congruence ticks are drawn as explicit strokes, sized so
% that each prints at about 2.2pt half-length at the figure's own scale
% (Figs. 4-7/4-8/4-9, split out and enlarged, carry the book's own
% tick-to-segment ratio instead and so print nearer 3pt).
\tikzset{vlab/.append style={outer sep=3.2pt},
         slab/.append style={outer sep=3.2pt}}
